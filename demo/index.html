python ztestkit.py \
  --uri-file /private/request.txt \
  --rpc-url http://127.0.0.1:50233 \
  --rpc-cookie /private/rpc-cookie \
  --account-uuid YOUR_ACCOUNT \
  --txid YOUR_TRANSACTION \
  --network regtest \
  --canary-file /private/canaries.txt \
  --log /private/app.log \
  --report /private/report.json
